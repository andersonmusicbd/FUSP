function classifyRow(itemDescription) { if (!itemDescription) return { category: 'IGNORAR', reason: 'Linha Vazia' }; const text = itemDescription.toString().trim(); const lower = text.toLowerCase(); if (!lower || lower.length < 2) return { category: 'IGNORAR', reason: 'Sem texto relevante' }; // 1. Linhas de Resumo, Saldos e Cabeçalhos Genéricos const blacklistKeywords = [ 'saldo final', 'saldo anterior', 'saldo pós', 'rendimento financeiro', 'disponível p/ extensão', 'total recebido', 'total previsto', 'resumo do projeto', 'rubrica', 'descrição', 'descrição do item' ]; if (blacklistKeywords.some(kw => lower.includes(kw))) { return { category: 'IGNORAR', reason: 'Linha de Resumo / Saldo' }; } // Se for exatamente o cabeçalho "NOME", ignora a linha de título if (lower === 'nome' || lower === 'nomes') { return { category: 'IGNORAR', reason: 'Cabeçalho da Coluna' }; } // 2. Equipamentos e Material Permanente const permanenteKeywords = ['permanente', 'equipamento', 'notebook', 'computador', 'servidor', 'máquina', 'hardware']; const mpRegex = /(?:^|[\s\-_])mp[1-6]?(?:$|[\s\-_]|\b)/i; if (permanenteKeywords.some(kw => lower.includes(kw)) || mpRegex.test(lower)) { return { category: 'Equipamento e Material Permanente', reason: 'Material Permanente' }; } // 3. Material de Consumo const consumoKeywords = ['material de consumo', 'reagente', 'insumos', 'suprimentos', 'almoxarifado', 'consumo']; const mcRegex = /(?:^|[\s\-_])mc[1-6]?(?:$|[\s\-_]|\b)/i; if (consumoKeywords.some(kw => lower.includes(kw)) || mcRegex.test(lower)) { return { category: 'Material de Consumo', reason: 'Material de Consumo' }; } // 4. Passagens, Diárias, Obras, Protótipos e Terceiros if (['passagem', 'passagens', 'voo', 'aéreo'].some(kw => lower.includes(kw))) return { category: 'Passagens', reason: 'Passagens' }; if (['diária', 'diarias', 'hospedagem', 'hotel', 'transporte'].some(kw => lower.includes(kw))) return { category: 'Diárias', reason: 'Diárias' }; if (['protótipo', 'prototipo', 'mockup'].some(kw => lower.includes(kw))) return { category: 'Protótipo', reason: 'Protótipo' }; if (['obra', 'obras', 'instalação', 'reforma'].some(kw => lower.includes(kw))) return { category: 'Obras e Instalações', reason: 'Obras' }; if (['terceiro', 'terceiros', 'pj', 'consultoria', 'serviço'].some(kw => lower.includes(kw))) return { category: 'Serviços de Terceiros / Outras Despesas', reason: 'Serviços' }; // 5. REGRA PADRÃO PARA NOMES E EQUIPE: // Qualquer item que sobrou (ex: "Zehbour Panossian", "CLT - Juliana", "PC1 - Sandro", "IC1") vai direto para Equipe. return { category: 'Equipe', reason: 'Identificado como Nome / Integrante da Equipe' }; }
